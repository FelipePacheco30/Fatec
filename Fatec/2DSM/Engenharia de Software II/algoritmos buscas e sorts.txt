busca sentinela
    sequential_ws(e:T, v:T[]):number{
        let pos:number = 0;
        v.push(e);
        while(v[pos]!= e)
            ++pos;
        if(pos == (v.length-1)){
            v.pop();
            return -1;
        } else{
            v.pop();
            return pos;
        }                     
    }
